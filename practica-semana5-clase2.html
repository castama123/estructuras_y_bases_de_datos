Práctica guiada en Replit: mide el problema N+1 y compruébalo con la solución

Vas a crear en la nube un pequeño programa Python con SQLAlchemy y SQLite que cuenta las consultas que envía cada escenario de SoundFlow, para que veas el N+1 en números y verifiques cuánto lo reduce la carga anticipada. No necesitas instalar nada en tu computador.

Paso a paso en Replit

  1. Entra a replit.com y crea una cuenta gratuita (puedes usar tu cuenta de Google).
  2. Crea un proyecto nuevo, elige la plantilla Python y ponle el nombre soundflow-nplus1.
  3. Abre la pestaña Shell y escribe pip install sqlalchemy. Espera a que termine.
  4. Abre el archivo main.py, borra lo que tenga y pega todo el código de abajo.
  5. Pulsa Run y mira los números que aparecen en la consola.

El código completo

Usa SQLite en memoria, así que no instalas MySQL. Cada escenario se ejecuta dos veces: primero sin optimizar (el problema) y luego optimizado (la solución), y al final imprime cuántas consultas se enviaron.

main.py 
from sqlalchemy import create_engine, Column, Integer, String, ForeignKey, Table, event
from sqlalchemy.orm import declarative_base, relationship, sessionmaker, joinedload, selectinload

# ---------- 1. Base de datos en memoria (SQLite): no hay que instalar nada ----------
engine = create_engine("sqlite:///:memory:")
Session = sessionmaker(bind=engine)
Base = declarative_base()

# ---------- 2. Contador de consultas ----------
# Cada vez que SQLAlchemy envía un SQL a la base de datos, este "espía" lo cuenta.
estado = {"consultas": 0, "mostrar_sql": False}

@event.listens_for(engine, "before_cursor_execute")
def contar_consultas(conn, cursor, statement, parameters, context, executemany):
    estado["consultas"] += 1
    if estado["mostrar_sql"]:
        texto = " ".join(statement.split())
        print("   SQL #" + str(estado["consultas"]) + ": SELECT ... " + texto[texto.find("FROM"):][:75] + "  " + str(parameters))

# ---------- 3. Modelos (las mismas tablas de SoundFlow de la clase) ----------
tbl_artistas_similares = Table(
    "tbl_artistas_similares", Base.metadata,
    Column("id_artista", Integer, ForeignKey("tbl_artistas.id_artista"), primary_key=True),
    Column("id_artista_similar", Integer, ForeignKey("tbl_artistas.id_artista"), primary_key=True),
)

class Artista(Base):
    __tablename__ = "tbl_artistas"
    id_artista = Column(Integer, primary_key=True)
    nombre_artista = Column(String(100))
    albumes = relationship("Album", back_populates="artista")
    similares = relationship(
        "Artista", secondary=tbl_artistas_similares,
        primaryjoin=id_artista == tbl_artistas_similares.c.id_artista,
        secondaryjoin=id_artista == tbl_artistas_similares.c.id_artista_similar,
    )

class Album(Base):
    __tablename__ = "tbl_albumes"
    id_album = Column(Integer, primary_key=True)
    titulo_album = Column(String(150))
    id_artista = Column(Integer, ForeignKey("tbl_artistas.id_artista"))
    artista = relationship("Artista", back_populates="albumes")
    canciones = relationship("Cancion", back_populates="album")

class Cancion(Base):
    __tablename__ = "tbl_canciones"
    id_cancion = Column(Integer, primary_key=True)
    titulo = Column(String(150))
    reproducciones = Column(Integer)
    id_album = Column(Integer, ForeignKey("tbl_albumes.id_album"))
    album = relationship("Album", back_populates="canciones")

Base.metadata.create_all(engine)

# ---------- 4. Datos de prueba ----------
# 8 artistas, cada uno con 3 álbumes; cada álbum con 3 canciones.
def cargar_datos():
    s = Session()
    artistas = [Artista(nombre_artista="Artista " + str(i)) for i in range(1, 9)]
    s.add_all(artistas)
    s.flush()
    albumes = []
    for a in artistas:
        for j in range(1, 4):
            albumes.append(Album(titulo_album="Álbum " + str(a.id_artista) + "." + str(j), artista=a))
    s.add_all(albumes)
    s.flush()
    n = 1
    for ronda in range(3):              # reparte las canciones álbum por álbum
        for alb in albumes:
            s.add(Cancion(titulo="Canción " + str(n), reproducciones=1000 * n, album=alb))
            n += 1
    artistas[0].similares = artistas[1:6]   # el Artista 1 tiene 5 similares (Artistas 2 a 6)
    s.commit()
    s.close()

cargar_datos()

def medir(titulo):
    print()
    print("=== " + titulo + " ===")
    estado["consultas"] = 0

# ---------- ESCENARIO 1: la playlist de 20 canciones y el álbum de cada una ----------
def escenario1_problema():
    medir("Escenario 1 SIN optimizar (carga perezosa)")
    s = Session()
    playlist = s.query(Cancion).order_by(Cancion.id_cancion).limit(20).all()   # 1 consulta
    for c in playlist:
        nombre_album = c.album.titulo_album       # 1 consulta MÁS por cada canción
    print("Consultas ejecutadas:", estado["consultas"])
    s.close()

def escenario1_solucion():
    medir("Escenario 1 CON joinedload")
    s = Session()
    playlist = (s.query(Cancion).options(joinedload(Cancion.album))
                .order_by(Cancion.id_cancion).limit(20).all())
    for c in playlist:
        nombre_album = c.album.titulo_album
    print("Consultas ejecutadas:", estado["consultas"])
    s.close()

# ---------- ESCENARIO 2: artistas similares y sus canciones (N+1 anidado) ----------
def escenario2_problema():
    medir("Escenario 2 SIN optimizar (N+1 anidado)")
    s = Session()
    principal = s.query(Artista).filter_by(nombre_artista="Artista 1").one()   # 1 consulta
    for sim in principal.similares:                  # 1 consulta para traer los similares
        for alb in sim.albumes:                      # 1 consulta por cada similar
            canciones = alb.canciones                # 1 consulta por cada álbum
    print("Consultas ejecutadas:", estado["consultas"])
    s.close()

def escenario2_solucion():
    medir("Escenario 2 CON selectinload encadenado")
    s = Session()
    principal = (s.query(Artista)
                 .options(selectinload(Artista.similares)
                          .selectinload(Artista.albumes)
                          .selectinload(Album.canciones))
                 .filter_by(nombre_artista="Artista 1").one())
    for sim in principal.similares:
        for alb in sim.albumes:
            canciones = alb.canciones
    print("Consultas ejecutadas:", estado["consultas"])
    s.close()

escenario1_problema()
escenario1_solucion()
escenario2_problema()
escenario2_solucion()

Ejercicios: cambia una cosa y comprueba

  1. En el escenario 1 sin optimizar, cambia limit(20) por limit(10) y ejecuta. Antes de correrlo, predice con 1 + N cuántas consultas saldrán. Deberías ver 11.
  2. En el escenario 1 optimizado, cambia joinedload por selectinload y ejecuta. Deberías ver 2 consultas, en vez de 1. Eso es porque selectinload hace una consulta aparte para los álbumes, en vez de un JOIN.
  3. Más similares: busca artistas[1:6] en el código y cámbialo por artistas[1:8]. Ejecuta y compara los dos números del escenario 2: el sin optimizar sube a 30, y el optimizado se queda en 4.
